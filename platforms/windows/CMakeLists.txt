# LiveWall for Windows.
#
# The whole dependency list is "the Windows SDK". Everything linked below ships
# with it, so there is no vcpkg manifest, no NuGet restore and no submodule —
# `cmake --preset default` on a machine with Visual Studio is the entire setup.
#
# That is not minimalism for its own sake. The macOS app is a 504 KB binary that
# idles at 12 MB, and a port that pulled in a media framework or a UI toolkit
# would lose the only property the project is actually about.
cmake_minimum_required(VERSION 3.21)

project(LiveWall
    VERSION 1.0.1
    DESCRIPTION "Low-power live wallpaper for Windows"
    LANGUAGES CXX)

if(NOT WIN32)
    message(FATAL_ERROR "platforms/windows targets Windows. For macOS use platforms/macos.")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(cmake/LiveWallDefaults.cmake)

# ---------------------------------------------------------------------------
# Core library
#
# Everything except the entry point, so the test executable can link the same
# objects the app runs. The logic worth testing — coverage geometry, frame
# pacing, output sizing, index decoding — is the same four areas the macOS
# suite covers, and none of it needs a display or a decoder.
# ---------------------------------------------------------------------------
add_library(livewall_core STATIC
    src/support/Log.cpp
    src/support/Strings.cpp
    src/support/Json.cpp
    src/support/Guid.cpp
    src/support/Footprint.cpp
    src/support/Paths.cpp
    src/support/Settings.cpp
    src/support/DesktopVisibility.cpp
    src/support/PowerMonitor.cpp
    src/support/StartupItem.cpp

    src/render/FitMode.cpp
    src/render/D3DDevice.cpp
    src/render/DesktopHost.cpp
    src/render/SwapChainTarget.cpp
    src/render/GradientSource.cpp
    src/render/VideoSource.cpp

    src/import/FramePacer.cpp
    src/import/CodecSupport.cpp
    src/import/Transcoder.cpp
    src/import/Library.cpp

    src/app/MonitorController.cpp
    src/app/WallpaperEngine.cpp
    src/app/TrayMenu.cpp
    src/app/AppHost.cpp
)

# res/ is on the include path so that resource.h — the one place menu command
# ids are defined — is shared by the .rc and the C++ that handles WM_COMMAND.
target_include_directories(livewall_core PUBLIC src res)
livewall_set_target_defaults(livewall_core)

# Every one of these is an import library from the Windows SDK.
target_link_libraries(livewall_core PUBLIC
    d3d11              # rendering
    dxgi               # swap chains, adapter/output enumeration
    dxguid             # the IIDs the two above are addressed by
    dcomp              # per-pixel alpha over the system wallpaper
    mf                 # Media Foundation core
    mfplat
    mfreadwrite        # IMFSourceReader / IMFSinkWriter
    mfuuid
    ole32
    oleaut32
    shlwapi
    shell32            # Shell_NotifyIcon, SHQueryUserNotificationState
    user32
    gdi32
    advapi32           # registry, for the Run key
    comdlg32           # the file picker
    wtsapi32           # session lock/unlock notifications
    powrprof
    psapi              # working set / private bytes
    winmm
)

# ---------------------------------------------------------------------------
# The app
#
# WIN32_EXECUTABLE gives it no console window. `--convert` still writes to
# stderr, which is inherited when it is started from a terminal, so the headless
# path behaves the same way the macOS one does.
# ---------------------------------------------------------------------------
add_executable(LiveWall WIN32
    src/main.cpp
    res/LiveWall.rc
)

target_link_libraries(LiveWall PRIVATE livewall_core)
livewall_set_target_defaults(LiveWall)

# The icons are generated by tools/make-icon.ps1 rather than committed, so a
# fresh clone has none. The resource compiler is not a C preprocessor — it has
# no __has_include and a missing file is a hard error — so the presence check
# happens here and arrives as a define.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/res/AppIcon.ico")
    target_compile_definitions(LiveWall PRIVATE LIVEWALL_HAVE_APPICON)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/res/TrayIcon.ico")
    target_compile_definitions(LiveWall PRIVATE LIVEWALL_HAVE_TRAYICON)
else()
    message(STATUS "No res/TrayIcon.ico — the tray will use a system icon. "
                   "Draw them with tools/make-icon.ps1.")
endif()

# The manifest declares per-monitor-v2 DPI awareness and Windows 10/11
# compatibility. Without the compatibility GUIDs, GetVersionEx and — more to the
# point — the DWM behaviours this app depends on report Windows 8 values.
target_link_options(LiveWall PRIVATE
    "/MANIFEST:NO"                       # the .rc embeds ours instead
)

# Pinned per configuration rather than left to the generator: a multi-config
# generator otherwise appends Debug/ or Release/, and every script and CI step
# that wants to find the executable would have to know which generator ran.
set_target_properties(LiveWall PROPERTIES
    OUTPUT_NAME "LiveWall"
    RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}"
)

# ---------------------------------------------------------------------------
# Shaders
#
# Compiled at build time into headers rather than at runtime. The macOS app
# measured runtime shader compilation at ~97 MB of resident graphics memory that
# is never released — more than the rest of the app combined — and D3D's
# D3DCompile has the same shape of cost. fxc ships with the Windows SDK.
# ---------------------------------------------------------------------------
include(cmake/CompileShaders.cmake)

livewall_add_shaders(livewall_core
    SOURCE  src/render/shaders/wallpaper.hlsl
    OUTPUT  ${CMAKE_BINARY_DIR}/generated/shaders
    ENTRIES
        VS_Fullscreen:vs_5_0
        PS_NV12:ps_5_0
        PS_P010:ps_5_0
        PS_Gradient:ps_5_0
)

target_include_directories(livewall_core PUBLIC ${CMAKE_BINARY_DIR}/generated)

# ---------------------------------------------------------------------------
# Tests
# ---------------------------------------------------------------------------
option(LIVEWALL_BUILD_TESTS "Build the LiveWall test suite" ON)

if(LIVEWALL_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()

message(STATUS "LiveWall ${PROJECT_VERSION} — tests: ${LIVEWALL_BUILD_TESTS}")
